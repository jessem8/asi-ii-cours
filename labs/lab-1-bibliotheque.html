Lab 1 — Bibliothèque

Un vers plusieurs dans les deux sens, et plusieurs vers plusieurs dans les deux sens. Ouvre la solution quand tu bloques.

60:00

Énoncé

Travail à faire. Créer une application Spring Boot (Maven, Java 21) avec Spring Web, Spring Data JPA, MySQL Driver et Lombok. Base bibliodb, port 8089, context path /biblio.

Entités, package entity, Lombok sur chaque classe :

Bibliotheque (idBibliotheque Long généré, nom, adresse).

Livre (idLivre Long généré, titre, genre de type enum Genre { ROMAN, ESSAI, POESIE }, datePublication de type Date).

Auteur (idAuteur Long généré, nom, prenom).

Associations :

Une bibliothèque contient plusieurs livres. Bidirectionnelle. Le livre est le fils.

Un livre est écrit par plusieurs auteurs et un auteur écrit plusieurs livres. Bidirectionnelle. L'auteur est le fils.

Package repository, package service (interface + implémentation), package control. CRUD Postman pour chaque entité.

1 — * entre Bibliotheque et Livre. * — * entre Livre et Auteur. Livre et Auteur sont les fils.

Checklist

Indices

Le fils Livre porte @ManyToOne Bibliotheque bibliotheque. Bibliotheque porte @OneToMany(mappedBy = "bibliotheque") et @JsonIgnore.

Le fils Auteur porte @ManyToMany(mappedBy = "auteurs") et @JsonIgnore. Livre porte @ManyToMany Set auteurs.

genre est un enum STRING. datePublication est @Temporal DATE. La colonne de la date est date_publication.

Tables attendues

  • bibliotheque (id_bibliotheque, nom, adresse)
  • livre (id_livre, titre, genre, date_publication, bibliotheque_id_bibliotheque)
  • auteur (id_auteur, nom, prenom)
  • livre_auteurs (livres_id_livre, auteurs_id_auteur)

MySQL n'était pas démarré quand ces pages ont été écrites. Cette liste suit la règle déjà vue dans tpfoyerdb : idChambre devient id_chambre, typeC devient typec. Les cinq projets compilent avec le JDK 21.

Solution

Même forme que le projet du cours. Tu tapes la version ci-dessous, avec le port 8089.

spring.datasource.url=jdbc:mysql://localhost:3306/bibliodb?createDatabaseIfNotExist=true
spring.datasource.username=root
spring.datasource.password=<the MAMP MySQL password>
spring.jpa.show-sql=true
spring.jpa.hibernate.ddl-auto=update
spring.jackson.deserialization.fail-on-null-for-primitives=false
server.port=8089
server.servlet.context-path=/biblio

import jakarta.persistence.*;
import lombok.*;
import com.fasterxml.jackson.annotation.JsonIgnore;

@Entity
@Getter
@Setter
@AllArgsConstructor
@NoArgsConstructor

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long idAuteur;
    private String nom;
    private String prenom;
    @ManyToMany(mappedBy = "auteurs")
    @JsonIgnore
    private Set<Livre> livres;
}

package tn.esprit.biblio.entity;

import jakarta.persistence.*;
import lombok.*;
import java.util.Date;
import java.util.Set;

@Entity
@Getter
@Setter
@AllArgsConstructor
@NoArgsConstructor
public class Livre {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long idLivre;
    private String titre;
    @Enumerated(EnumType.STRING)
    private Genre genre;
    @Temporal(TemporalType.DATE)
    private Date datePublication;
    @ManyToOne
    private Bibliotheque bibliotheque;
    @ManyToMany
    private Set<Auteur> auteurs;
}

package tn.esprit.biblio.service;

import lombok.AllArgsConstructor;
import org.springframework.stereotype.Service;
import tn.esprit.biblio.entity.Auteur;
import tn.esprit.biblio.repository.AuteurRepository;

import java.util.List;

@Service
@AllArgsConstructor
public class AuteurServiceImpl implements IAuteurService {

    AuteurRepository auteurRepository;

    public List<Auteur> retrieveAllAuteurs() {
        return auteurRepository.findAll();
    }

    public Auteur retrieveAuteur(Long auteurId) {
        return auteurRepository.findById(auteurId).get();
    }

    public Auteur addAuteur(Auteur auteur) {
        return auteurRepository.save(auteur);
    }

    public void removeAuteur(Long auteurId) {
        auteurRepository.deleteById(auteurId);
    }

    public Auteur modifyAuteur(Auteur auteur) {
        return auteurRepository.save(auteur);
    }
}

package tn.esprit.biblio.control;

import lombok.AllArgsConstructor;
import org.springframework.web.bind.annotation.*;
import tn.esprit.biblio.entity.Auteur;
import tn.esprit.biblio.service.IAuteurService;

import java.util.List;

@RestController
@AllArgsConstructor
@RequestMapping("/auteur")
public class AuteurRestController {

    IAuteurService auteurService;

    // http://localhost:8089/biblio/auteur/retrieve-all-auteurs
    @GetMapping("/retrieve-all-auteurs")
    public List<Auteur> getAuteurs() {
        return auteurService.retrieveAllAuteurs();
    }

    // http://localhost:8089/biblio/auteur/retrieve-auteur/1
    @GetMapping("/retrieve-auteur/{auteur-id}")
    public Auteur retrieveAuteur(@PathVariable("auteur-id") Long id) {
        return auteurService.retrieveAuteur(id);
    }

    @PostMapping("/add-auteur")
    public Auteur addAuteur(@RequestBody Auteur body) {
        return auteurService.addAuteur(body);
    }

    @DeleteMapping("/remove-auteur/{auteur-id}")
    public void removeAuteur(@PathVariable("auteur-id") Long id) {
        auteurService.removeAuteur(id);
    }

    @PutMapping("/modify-auteur")
    public Auteur modifyAuteur(@RequestBody Auteur body) {
        return auteurService.modifyAuteur(body);
    }
}
