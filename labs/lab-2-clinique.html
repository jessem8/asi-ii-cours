Lab 2 — Clinique

Un vers plusieurs dans les deux sens, et plusieurs vers plusieurs dans un seul sens. Le boolean disponible doit être dans le JSON.

60:00

Énoncé

Travail à faire. Même socle technique. Base cliniquedb, port 8089, context path /clinique.

Clinique (idClinique Long généré, nom, adresse).

Medecin (idMedecin Long généré, nom, prenom, specialite enum Specialite { CARDIO, PNEUMO, DERMATO }, disponible boolean).

Patient (idPatient Long généré, nom, prenom, cin Long).

Une clinique emploie plusieurs médecins. Bidirectionnelle. Le médecin est le fils.

Un médecin suit plusieurs patients, un patient peut être suivi par plusieurs médecins. Unidirectionnelle, seulement depuis Medecin.

Repository, service, controller, Postman.

1 — * dans les deux sens. * — * seulement depuis Medecin. Pas de champ sur Patient.

Checklist

Indices

Pas de mappedBy sur patients. @ManyToMany Set Patient patients sur Medecin seulement. Cela crée une join table.

Medecin est le fils de Clinique : @ManyToOne Clinique clinique, et @JsonIgnore sur la collection de Clinique.

Tables attendues

  • clinique (id_clinique, nom, adresse)
  • medecin (id_medecin, nom, prenom, specialite, disponible, clinique_id_clinique)
  • patient (id_patient, nom, prenom, cin)
  • medecin_patients (medecin_id_medecin, patients_id_patient)

MySQL n'était pas démarré quand ces pages ont été écrites. Cette liste suit la règle déjà vue dans tpfoyerdb : idChambre devient id_chambre, typeC devient typec. Les cinq projets compilent avec le JDK 21.

Solution

Même forme que le projet du cours. Tu tapes la version ci-dessous, avec le port 8089.

spring.datasource.url=jdbc:mysql://localhost:3306/cliniquedb?createDatabaseIfNotExist=true
spring.datasource.username=root
spring.datasource.password=<the MAMP MySQL password>
spring.jpa.show-sql=true
spring.jpa.hibernate.ddl-auto=update
spring.jackson.deserialization.fail-on-null-for-primitives=false
server.port=8089
server.servlet.context-path=/clinique

package tn.esprit.clinique.entity;

import jakarta.persistence.*;
import lombok.*;
import com.fasterxml.jackson.annotation.JsonIgnore;
import java.util.Set;

@Entity
@Getter
@Setter
@AllArgsConstructor
@NoArgsConstructor
public class Clinique {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long idClinique;
    private String nom;
    private String adresse;
    @OneToMany(mappedBy = "clinique")
    @JsonIgnore
    private Set<Medecin> medecins;
}

package tn.esprit.clinique.entity;

import jakarta.persistence.*;
import lombok.*;
import java.util.Set;

@Entity
@Getter
@Setter
@AllArgsConstructor
@NoArgsConstructor
public class Medecin {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long idMedecin;
    private String nom;
    private String prenom;
    @Enumerated(EnumType.STRING)
    private Specialite specialite;
    private boolean disponible;
    @ManyToOne
    private Clinique clinique;
    @ManyToMany
    private Set<Patient> patients;
}

package tn.esprit.clinique.service;

import lombok.AllArgsConstructor;
import org.springframework.stereotype.Service;
import tn.esprit.clinique.entity.Clinique;
import tn.esprit.clinique.repository.CliniqueRepository;

import java.util.List;

@Service
@AllArgsConstructor
public class CliniqueServiceImpl implements ICliniqueService {

    CliniqueRepository cliniqueRepository;

    public List<Clinique> retrieveAllCliniques() {
        return cliniqueRepository.findAll();
    }

    public Clinique retrieveClinique(Long cliniqueId) {
        return cliniqueRepository.findById(cliniqueId).get();
    }

    public Clinique addClinique(Clinique clinique) {
        return cliniqueRepository.save(clinique);
    }

    public void removeClinique(Long cliniqueId) {
        cliniqueRepository.deleteById(cliniqueId);
    }

    public Clinique modifyClinique(Clinique clinique) {
        return cliniqueRepository.save(clinique);
    }
}

package tn.esprit.clinique.control;

import lombok.AllArgsConstructor;
import org.springframework.web.bind.annotation.*;
import tn.esprit.clinique.entity.Clinique;
import tn.esprit.clinique.service.ICliniqueService;

import java.util.List;

@RestController
@AllArgsConstructor
@RequestMapping("/clinique")
public class CliniqueRestController {

    ICliniqueService cliniqueService;

    // http://localhost:8089/clinique/clinique/retrieve-all-cliniques
    @GetMapping("/retrieve-all-cliniques")
    public List<Clinique> getCliniques() {
        return cliniqueService.retrieveAllCliniques();
    }

    // http://localhost:8089/clinique/clinique/retrieve-clinique/1
    @GetMapping("/retrieve-clinique/{clinique-id}")
    public Clinique retrieveClinique(@PathVariable("clinique-id") Long id) {
        return cliniqueService.retrieveClinique(id);
    }

    @PostMapping("/add-clinique")
    public Clinique addClinique(@RequestBody Clinique body) {
        return cliniqueService.addClinique(body);
    }

    @DeleteMapping("/remove-clinique/{clinique-id}")
    public void removeClinique(@PathVariable("clinique-id") Long id) {
        cliniqueService.removeClinique(id);
    }

    @PutMapping("/modify-clinique")
    public Clinique modifyClinique(@RequestBody Clinique body) {
        return cliniqueService.modifyClinique(body);
    }
}
