Lab 5 — Examen blanc

Lis le sujet. Lance le chrono. Les indices et la solution restent fermés jusqu'à 00:00.

60:00

Énoncé

Examen blanc. 60 minutes. Base stagesdb, port 8089, context path /stages. Trois entités, deux associations, repository, service, controller, un CRUD Postman.

Entreprise (idEntreprise Long généré, nom, adresse).

Offre (idOffre Long généré, titre, typeO enum TypeOffre { STAGE, ALTERNANCE, CDI }, dateDebut Date, publiee boolean).

Candidat (idCandidat Long généré, nom, prenom).

Une entreprise publie plusieurs offres. Bidirectionnelle. L'offre est le fils.

Une offre concerne plusieurs candidats, un candidat répond à plusieurs offres. Bidirectionnelle. Le candidat est le fils.

Le champ typeO se termine par une majuscule. Rappelle-toi la colonne de typeC dans tpfoyerdb.

1 — * , Offre est le fils. * — * , Candidat est le fils.

import jakarta.persistence.*;

    private Long idCandidat;

    private String prenom;
    @ManyToMany(mappedBy = "candidats")

package tn.esprit.stages.entity;

import jakarta.persistence.*;
import lombok.*;
import java.util.Date;
import java.util.Set;

@Entity
@Getter
@Setter
@AllArgsConstructor
@NoArgsConstructor
public class Offre {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long idOffre;
    private String titre;
    @Enumerated(EnumType.STRING)
    private TypeOffre typeO;
    @Temporal(TemporalType.DATE)
    private Date dateDebut;
    private boolean publiee;
    @ManyToOne
    private Entreprise entreprise;
    @ManyToMany
    private Set<Candidat> candidats;
}

package tn.esprit.stages.service;

import lombok.AllArgsConstructor;
import org.springframework.stereotype.Service;
import tn.esprit.stages.entity.Candidat;
import tn.esprit.stages.repository.CandidatRepository;

import java.util.List;

@Service
@AllArgsConstructor
public class CandidatServiceImpl implements ICandidatService {

    CandidatRepository candidatRepository;

    public List<Candidat> retrieveAllCandidats() {
        return candidatRepository.findAll();
    }

    public Candidat retrieveCandidat(Long candidatId) {
        return candidatRepository.findById(candidatId).get();
    }

    public Candidat addCandidat(Candidat candidat) {
        return candidatRepository.save(candidat);
    }

    public void removeCandidat(Long candidatId) {
        candidatRepository.deleteById(candidatId);
    }

    public Candidat modifyCandidat(Candidat candidat) {
        return candidatRepository.save(candidat);
    }
}

package tn.esprit.stages.control;

import lombok.AllArgsConstructor;
import org.springframework.web.bind.annotation.*;
import tn.esprit.stages.entity.Candidat;
import tn.esprit.stages.service.ICandidatService;

import java.util.List;

@RestController
@AllArgsConstructor
@RequestMapping("/candidat")
public class CandidatRestController {

    ICandidatService candidatService;

    // http://localhost:8089/stages/candidat/retrieve-all-candidats
    @GetMapping("/retrieve-all-candidats")
    public List<Candidat> getCandidats() {
        return candidatService.retrieveAllCandidats();
    }

    // http://localhost:8089/stages/candidat/retrieve-candidat/1
    @GetMapping("/retrieve-candidat/{candidat-id}")
    public Candidat retrieveCandidat(@PathVariable("candidat-id") Long id) {
        return candidatService.retrieveCandidat(id);
    }

    @PostMapping("/add-candidat")
    public Candidat addCandidat(@RequestBody Candidat body) {
        return candidatService.addCandidat(body);
    }

    @DeleteMapping("/remove-candidat/{candidat-id}")
    public void removeCandidat(@PathVariable("candidat-id") Long id) {
        candidatService.removeCandidat(id);
    }

    @PutMapping("/modify-candidat")
    public Candidat modifyCandidat(@RequestBody Candidat body) {
        return candidatService.modifyCandidat(body);
    }
}
